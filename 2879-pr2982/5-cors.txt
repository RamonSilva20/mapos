-- Origin: http://evil.test
$ -X POST http://127.0.0.1:8982/index.php/api/v1/client/auth -H Origin: http://evil.test -H Content-Type: application/json -d {"email":"cliente@mapos.test","senha":"errada"}

$ -X OPTIONS http://127.0.0.1:8982/index.php/api/v1/client/auth -H Origin: http://evil.test -H Access-Control-Request-Method: POST -H Access-Control-Request-Headers: content-type,authorization
